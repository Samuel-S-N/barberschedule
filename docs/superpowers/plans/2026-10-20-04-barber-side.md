# Barber Side Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans. Steps use checkbox (`- [ ]`) syntax.

**Goal:** (1) "Disponibilidade" menu in the barber profile (working hours, lunch break, absences) replacing the agenda's "Suas folgas" block; (2) clear (X) button on the client search; (3) clickable agenda cards with WhatsApp/phone; (4) "Salvar perfil" button at the end of Meus dados.

**Architecture:** Lunch break = two nullable columns on `barbers` (`lunch_start`, `lunch_end`) subtracted in `get_available_slots` as one more blocked range; edited through `set_my_lunch_break`. Absences reuse the existing `schedule_overrides(kind='block')` API, moved out of `my-agenda.tsx` into a new profile screen. Agenda RPC now returns the customer's phone for every customer so the card can show contact buttons.

**Tech Stack:** Postgres (pgTAP), Expo Router, React Query, NativeWind, Jest.

**Spec:** `docs/superpowers/plans/2026-10-20-ux-overhaul-index.md`

alter table public.barbers
  add column lunch_start time,
  add column lunch_end time,
  add constraint barbers_lunch_valid check (
    (lunch_start is null and lunch_end is null) or (lunch_start is not null and lunch_end is not null and lunch_start < lunch_end)
  );

create function public.get_my_lunch_break()
returns table (lunch_start time, lunch_end time)
language sql stable security definer set search_path = pg_catalog, public, pg_temp as $$
  select b.lunch_start, b.lunch_end from public.barbers b where b.user_id = auth.uid() and b.active;
$$;

create function public.set_my_lunch_break(new_start time, new_end time)
returns void language plpgsql security definer set search_path = pg_catalog, public, pg_temp as $$
begin
  if (new_start is null) <> (new_end is null) or (new_start is not null and new_start >= new_end) then
    raise exception using errcode = 'P0020', message = 'LUNCH_INVALID';
  end if;
  update public.barbers set lunch_start = new_start, lunch_end = new_end, updated_at = clock_timestamp()
  where user_id = auth.uid() and active;
  if not found then
    raise exception using errcode = 'P0019', message = 'BARBER_NOT_LINKED';
  end if;
end;
$$;

revoke all on function public.get_my_lunch_break() from public;
revoke all on function public.set_my_lunch_break(time, time) from public;
grant execute on function public.get_my_lunch_break() to authenticated;
grant execute on function public.set_my_lunch_break(time, time) to authenticated;
```

// api.ts (append) — uses existing throwBarberError
export async function getMyLunchBreak(supabase: Pick<SupabaseClient, "rpc">) {
  const { data, error } = await supabase.rpc("get_my_lunch_break");
  if (error) throwBarberError(error);
  const row = (Array.isArray(data) ? data[0] : data) as { lunch_end: string | null; lunch_start: string | null } | undefined;

  return row?.lunch_start && row.lunch_end ? { end: row.lunch_end.slice(0, 5), start: row.lunch_start.slice(0, 5) } : null;
}

export async function setMyLunchBreak(supabase: Pick<SupabaseClient, "rpc">, lunch: { end: string; start: string } | null) {
  const { error } = await supabase.rpc("set_my_lunch_break", { new_end: lunch?.end ?? null, new_start: lunch?.start ?? null });
  if (error) throwBarberError(error);
}
```
```ts
// absence-range.ts
const MAX_DAYS = 31;

export function expandAbsenceDates(startDate: string, endDate: string | null) {
  const last = endDate ?? startDate;
  const dates: string[] = [];

  for (let day = new Date(`${startDate}T12:00:00Z`); day <= new Date(`${last}T12:00:00Z`); day.setUTCDate(day.getUTCDate() + 1)) {
    dates.push(day.toISOString().slice(0, 10));
    if (dates.length > MAX_DAYS) break;
  }
  if (dates.length === 0 || dates.length > MAX_DAYS) throw new RangeError("ABSENCE_RANGE_INVALID");

  return dates;
}
```
