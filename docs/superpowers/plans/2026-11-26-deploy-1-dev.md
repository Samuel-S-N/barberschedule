# Deploy 1: dev pipeline, scheduled dispatch and Sentry Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Every merge to `main` deploys to the hosted dev environment (`zenc-dev` + the `zenc-web-dev` Worker), notifications are actually dispatched by a cron job, and errors reach Sentry from the app and the Edge Functions.

**Architecture:** One reusable workflow (`deploy-env.yml`) does `db push`, `config push`, `functions deploy`, web export, Sentry source maps and `wrangler deploy` for whichever GitHub Environment calls it; `deploy.yml` chains verify, then dev (plan 2 adds production). Hosted-only auth settings live in `[remotes.dev]` of `supabase/config.toml`. A new migration schedules `dispatch-notifications` with pg_cron + pg_net + Vault. Sentry is a thin scrubbed wrapper in the app and a dependency-free envelope client (or `@sentry/deno`, decided by a spike) in the functions.

**Tech Stack:** GitHub Actions, Supabase CLI 2.114, Cloudflare Workers (static assets, wrangler), pg_cron/pg_net/Vault, `@sentry/react-native` (Expo plugin), Jest, pgTAP.

**Spec:** `docs/superpowers/specs/2026-11-26-deploy-05-design.md` (sections 3 to 6 and 7 for the risks this plan spikes).

- Claude never types or sees a secret: every key, token and password is pasted by the user into GitHub, Supabase or Cloudflare (spec section 6). Commands that need them are run by the user or by CI.

export function buildShopUrl(template: string, slug: string, path = "/"): string {
  if (!template.includes("{slug}")) throw new Error("SHOP_URL_TEMPLATE must contain {slug}");

  const filled = template.replace("{slug}", slug);
  const queryAt = filled.indexOf("?");
  const base = queryAt === -1 ? filled : filled.slice(0, queryAt);
  const query = queryAt === -1 ? "" : filled.slice(queryAt);

  return `${base.replace(/\/+$/, "")}${path.startsWith("/") ? path : `/${path}`}${query}`;
}
```

// Only the scheduler may drain the outbox; the anon key ships inside the app. Two credentials are accepted: the service-role key
// (manual calls) and DISPATCH_TOKEN (the pg_cron job), so the service-role key never has to sit in Vault or in GitHub.
// Compared in constant time so the response time does not leak how much of a key matched.
function matches(authorization: string | null, key: string) {
  const expected = `Bearer ${key}`;
  if (!authorization || authorization.length !== expected.length) return false;

  let diff = 0;
  for (let i = 0; i < expected.length; i++) diff |= authorization.charCodeAt(i) ^ expected.charCodeAt(i);
  return diff === 0;
}

export function isServiceRequest(authorization: string | null, ...keys: Array<string | undefined>) {
  let ok = false;
  for (const key of keys) if (key && matches(authorization, key)) ok = true;
  return ok;
}

export const dispatchToken = (value: string | undefined) => (value && value.length >= 32 ? value : undefined);
```

```sql
-- Scheduled dispatch of the notification outbox. 0022 and 0062 only enqueue rows with pg_cron; nothing called the Edge Function.
-- The URL and the token live in Vault (set per project: see docs/hosted-deploy.md), never in a migration. Without them (local
-- stack, CI, a project not configured yet) the function returns null and the job is a harmless no-op.
create or replace function public.invoke_dispatch_notifications()
returns bigint
language plpgsql
security definer
set search_path = pg_catalog, public, pg_temp
as $$
declare
  target_url text;
  token text;
  request_id bigint;
begin
  if to_regclass('vault.decrypted_secrets') is null or to_regprocedure('net.http_post(text,jsonb,jsonb,jsonb,integer)') is null then
    return null;
  end if;

  select decrypted_secret into target_url from vault.decrypted_secrets where name = 'dispatch_url';
  select decrypted_secret into token from vault.decrypted_secrets where name = 'dispatch_token';
  if target_url is null or token is null then
    return null;
  end if;

  select net.http_post(
    url := target_url,
    headers := jsonb_build_object('Content-Type', 'application/json', 'Authorization', 'Bearer ' || token),
    body := '{}'::jsonb,
    timeout_milliseconds := 20000
  ) into request_id;

  return request_id;
end;
$$;

revoke all on function public.invoke_dispatch_notifications() from public, anon, authenticated;

do $schedule$
begin
  if exists (select 1 from pg_available_extensions where name = 'pg_net') then
    execute 'create extension if not exists pg_net with schema extensions';
  end if;
  if exists (select 1 from pg_available_extensions where name = 'pg_cron') then
    execute 'create extension if not exists pg_cron with schema extensions';
    if to_regnamespace('cron') is not null then
      execute $sql$select cron.schedule('zenc-dispatch-notifications', '* * * * *', 'select public.invoke_dispatch_notifications();')$sql$;
    end if;
  end if;
end
$schedule$;
```

```ts
// Pure and dependency-free: imported by the app (Metro) and by the Edge Functions (Deno). Sentry gets errors, never personal data.
type Json = { [key: string]: unknown };

const EMAIL = /[^\s@"'<>]+@[^\s@"'<>]+\.[^\s@"'<>]+/g;
const withoutQuery = (url: string) => url.replace(/[?#].*$/, "");

export const scrubText = (text: string) => text.replace(EMAIL, "[email]");

export function scrubBreadcrumb<T>(crumb: T): T {
  const copy = { ...(crumb as Json) };

  if (typeof copy.message === "string") copy.message = scrubText(copy.message);
  if (copy.data && typeof copy.data === "object") {
    const data = { ...(copy.data as Json) };

    if (typeof data.url === "string") data.url = withoutQuery(data.url);
    copy.data = data;
  }

  return copy as T;
}

export function scrubEvent<T>(event: T): T {
  const copy = { ...(event as Json) };
  const request = copy.request as Json | undefined;
  const exception = copy.exception as { values?: Json[] } | undefined;

  delete copy.user;
  if (request) copy.request = { ...(request.method ? { method: request.method } : {}), ...(typeof request.url === "string" ? { url: withoutQuery(request.url) } : {}) };
  if (typeof copy.message === "string") copy.message = scrubText(copy.message);
  if (exception?.values) copy.exception = { ...exception, values: exception.values.map((entry) => ({ ...entry, value: typeof entry.value === "string" ? scrubText(entry.value) : entry.value })) };
  if (Array.isArray(copy.breadcrumbs)) copy.breadcrumbs = copy.breadcrumbs.map((crumb) => scrubBreadcrumb(crumb));

  return copy as T;
}
```

```ts
import { scrubEvent, scrubText } from "./scrub.ts";

type Send = (url: string, init: { body: string; headers: Record<string, string>; method: string }) => Promise<Response>;

export function parseDsn(dsn: string | undefined): { host: string; key: string; project: string } | null {
  try {
    const url = new URL(dsn ?? "");
    const project = url.pathname.replace(/^\/+/, "");

    return url.username && project ? { host: url.host, key: url.username, project } : null;
  } catch {
    return null;
  }
}

// Errors only, no SDK: one POST of one envelope per error (Sentry's documented envelope format). Reporting must never break the
// function it observes, so every failure is swallowed.
export function createReporter({ dsn, environment, fetch: send = fetch as Send, id = () => crypto.randomUUID().replaceAll("-", ""), now = () => new Date() }: {
  dsn: string | undefined;
  environment: string;
  fetch?: Send;
  id?: () => string;
  now?: () => Date;
}) {
  const target = parseDsn(dsn);

  return async (error: unknown, tags: Record<string, string> = {}): Promise<void> => {
    if (!target) return;
    try {
      const eventId = id();
      const err = error instanceof Error ? error : new Error(String(error));
      const event = scrubEvent({
        environment,
        event_id: eventId,
        exception: { values: [{ type: err.name, value: err.message }] },
        extra: { stack: scrubText(err.stack ?? "") },
        level: "error",
        platform: "javascript",
        tags,
        timestamp: now().toISOString(),
      });
      const body = [JSON.stringify({ event_id: eventId, sent_at: now().toISOString() }), JSON.stringify({ type: "event" }), JSON.stringify(event)].join("\n");

      await send(`https://${target.host}/api/${target.project}/envelope/`, {
        body,
        headers: { "Content-Type": "application/x-sentry-envelope", "X-Sentry-Auth": `Sentry sentry_version=7, sentry_key=${target.key}` },
        method: "POST",
      });
    } catch {
      // Sentry down or blocked: nothing to do.
    }
  };
}

export const withMonitoring = (handler: (request: Request) => Promise<Response> | Response, report: ReturnType<typeof createReporter>) =>
  async (request: Request): Promise<Response> => {
    try {
      return await handler(request);
    } catch (error) {
      await report(error);

      return Response.json({ code: "INTERNAL_ERROR" }, { status: 500 });
    }
  };
```
