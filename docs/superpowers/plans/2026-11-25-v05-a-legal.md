# 0.5 Part A: Legal texts Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Three public legal documents (privacy policy, terms of use, business agreement) with versions, company placeholders, consent records per document and a re-accept gate.

**Architecture:** The documents are typed data files rendered by one screen component; company data comes from one config file and renders highlighted when empty. Consents are recorded through a new RPC (the table is read-only for clients); a pure function decides which documents a user still has to accept, and `resolveAuthRedirect` sends them to `/accept-terms`.

**Tech Stack:** Expo Router, React Native Web, NativeWind, i18next (only for chrome text, the documents themselves are Portuguese data), Supabase (plpgsql, pgTAP), Jest, Playwright.

**Spec:** `docs/superpowers/specs/2026-11-25-v05-readiness-design.md` (section A and decisions 2, 4, 5, 10).

// Fill the four empty fields before the public launch (docs/hosted-deploy.md has the checklist item).

  dpoEmail: "privacidade@zencpro.com.br",
  endereco: "",
  foro: "",
  razaoSocial: "",
};

  cnpj: "[CNPJ]",
  dpoEmail: "[E-MAIL DO ENCARREGADO]",
  endereco: "[ENDEREÇO]",
  foro: "[FORO]",
  razaoSocial: "[RAZÃO SOCIAL]",
};

```ts
// src/features/legal/placeholders.ts
import { COMPANY, PLACEHOLDER_LABEL } from "./company";
import type { CompanyKey } from "./company";

export type Segment = { highlight: boolean; text: string };

const TOKEN = /\{\{(razaoSocial|cnpj|endereco|foro|dpoEmail)\}\}/g;

// "Foro: {{foro}}." -> plain "Foro: ", the value (highlighted placeholder while empty), plain ".".
export function splitSegments(text: string, company: Record<CompanyKey, string> = COMPANY): Segment[] {
  const segments: Segment[] = [];
  let last = 0;

  for (const match of text.matchAll(TOKEN)) {
    const key = match[1] as CompanyKey;
    const value = company[key].trim();

    if (match.index > last) segments.push({ highlight: false, text: text.slice(last, match.index) });
    segments.push(value ? { highlight: false, text: value } : { highlight: true, text: PLACEHOLDER_LABEL[key] });
    last = match.index + match[0].length;
  }
  if (last < text.length) segments.push({ highlight: false, text: text.slice(last) });

  return segments.length > 0 ? segments : [{ highlight: false, text }];
}
```

-- Legal documents have their own kinds and versions: the consents table grows a `business` kind, clients record consent through a
-- function (the table is read-only for them) and signup keeps one version per document.
alter table public.consents drop constraint consents_kind_check;
alter table public.consents add constraint consents_kind_check check (kind in ('terms', 'privacy', 'business'));

create function public.accept_legal_documents(p_documents jsonb)
returns void
language plpgsql
security definer
set search_path = pg_catalog, public, pg_temp
as $$
declare
  actor uuid := auth.uid();
  item jsonb;
  doc_kind text;
  doc_version text;
  is_owner boolean;
begin
  if actor is null then
    raise exception using errcode = '42501', message = 'FORBIDDEN';
  end if;
  if jsonb_typeof(p_documents) is distinct from 'array' or jsonb_array_length(p_documents) = 0 then
    raise exception using errcode = 'P0046', message = 'LEGAL_CONSENT_INVALID';
  end if;

  select role = 'owner' into is_owner from public.profiles where user_id = actor;

  for item in select * from jsonb_array_elements(p_documents) loop
    doc_kind := item ->> 'kind';
    doc_version := nullif(btrim(item ->> 'version'), '');

    if doc_kind is null or doc_kind not in ('terms', 'privacy', 'business')
      or doc_version is null
      or (doc_kind = 'business' and not coalesce(is_owner, false))
    then
      raise exception using errcode = 'P0046', message = 'LEGAL_CONSENT_INVALID';
    end if;

    insert into public.consents (user_id, kind, version) values (actor, doc_kind, doc_version) on conflict do nothing;
  end loop;
end;
$$;

create or replace function public.ensure_my_customer(p_shop_id uuid)
returns public.customers
language plpgsql
security definer
set search_path = pg_catalog, public, pg_temp
as $$
declare
  actor uuid := auth.uid();
  shop uuid;
  account_email text;
  account_confirmed boolean;
  meta jsonb;
  consent_version text;
  privacy_version text;
  result public.customers;
  mine public.customers;
  stale public.customers;
begin
  if actor is null
    or coalesce((select role::text from public.profiles where user_id = actor), '') <> 'customer'
  then
    raise exception using errcode = '42501', message = 'FORBIDDEN';
  end if;

  select id into shop from public.shops where id = p_shop_id;
  if shop is null then
    raise exception using errcode = 'P0007', message = 'CUSTOMER_UNAVAILABLE';
  end if;

  select email, email_confirmed_at is not null, coalesce(raw_user_meta_data, '{}'::jsonb)
  into account_email, account_confirmed, meta
  from auth.users where id = actor;

  if account_confirmed and (select email_claim_enabled from public.app_settings) then
    update public.customers c
    set user_id = actor
    where c.shop_id = shop
      and c.user_id is null
      and c.active
      and lower(c.email) = lower(account_email)
      and not exists (select 1 from public.customers x where x.shop_id = shop and x.user_id = actor);

    select * into mine from public.customers where shop_id = shop and user_id = actor;
    if found then
      select * into stale from public.customers
      where shop_id = shop and user_id is null and active and lower(email) = lower(account_email) and id <> mine.id;
      if found then
        update public.appointments set customer_id = mine.id where customer_id = stale.id;
        update public.recurrence_series set customer_id = mine.id where customer_id = stale.id;
        insert into public.professional_customer_notes (professional_id, customer_id, note, updated_at)
        select professional_id, mine.id, note, updated_at from public.professional_customer_notes where customer_id = stale.id
        on conflict do nothing;
        delete from public.customers where id = stale.id;
        update public.customers set email = coalesce(email, stale.email), phone = coalesce(phone, stale.phone) where id = mine.id;
      end if;
    end if;
  end if;

  insert into public.customers (shop_id, user_id, full_name, email, phone)
  values (
    shop,
    actor,
    coalesce((select full_name from public.profiles where user_id = actor), split_part(account_email, '@', 1)),
    case when exists (
      select 1 from public.customers x where x.shop_id = shop and lower(x.email) = lower(account_email)
    ) then null else account_email end,
    nullif(btrim(meta ->> 'phone'), '')
  )
  on conflict (shop_id, user_id) where user_id is not null do nothing;

  select * into result from public.customers where shop_id = shop and user_id = actor;

  consent_version := nullif(btrim(meta ->> 'accepted_terms_version'), '');
  privacy_version := coalesce(nullif(btrim(meta ->> 'accepted_privacy_version'), ''), consent_version);
  if consent_version is not null then

  end if;
  if privacy_version is not null then
    insert into public.consents (user_id, kind, version) values (actor, 'privacy', privacy_version) on conflict do nothing;
  end if;

  return result;
end;

// src/features/legal/pending.ts
import type { AppRole } from "../auth/types";
import { LEGAL_VERSIONS } from "./index";
import type { LegalKind } from "./types";

export type Consent = { kind: LegalKind; version: string };

// Owners accept the business agreement in the setup checklist (part C) the first time, so the gate only asks for it afterwards.
export function requiredKinds(role: AppRole | null, { setupComplete }: { setupComplete: boolean }): LegalKind[] {
  if (!role) return [];
  if (role === "owner" && setupComplete) return ["terms", "privacy", "business"];

  return ["terms", "privacy"];
}

// Versions are ISO dates, so a plain string comparison orders them.
export function pendingKinds(required: LegalKind[], consents: Consent[], versions: Record<LegalKind, string> = LEGAL_VERSIONS): LegalKind[] {
  return required.filter((kind) => !consents.some((consent) => consent.kind === kind && consent.version >= versions[kind]));
}

import type { LegalKind } from "./types";

export async function listMyConsents(supabase: Pick<SupabaseClient, "from">): Promise<Consent[]> {
  const { data, error } = await supabase.from("consents").select("kind, version");

  if (error) throw toDomainError(error);

import { useQuery } from "@tanstack/react-query";

import { useSupabaseSession } from "../../providers/AppProviders";
import { listMyConsents } from "./api";
import { pendingKinds, requiredKinds } from "./pending";
import type { LegalKind } from "./types";

// `setupComplete` is part C's flag (an owner whose business is not open yet accepts the business agreement in the checklist).
export function usePendingLegal(setupComplete = true): { isLoading: boolean; pending: LegalKind[] } {
  const { profile, session, supabase } = useSupabaseSession();
  const query = useQuery({
    enabled: Boolean(session) && Boolean(profile),
    queryFn: () => listMyConsents(supabase),
    queryKey: ["my-consents", session?.user.id],
    staleTime: 60_000,
  });

  if (!session || !profile) return { isLoading: false, pending: [] };
  if (query.isPending) return { isLoading: true, pending: [] };
  // A failed lookup must not lock the user out of the app: treat it as "nothing pending" and let the next refetch decide.
  if (query.isError) return { isLoading: false, pending: [] };

  return { isLoading: false, pending: pendingKinds(requiredKinds(profile.role, { setupComplete }), query.data) };
}
