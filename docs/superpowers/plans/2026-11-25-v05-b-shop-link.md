```ts
// src/features/shops/qr.ts
import QRCode from "qrcode";

// `create` builds the module matrix without a canvas or a DOM, so it works the same in Jest, Metro (native) and the web bundle.
export function qrMatrix(text: string): boolean[][] {
  const { modules } = QRCode.create(text, { errorCorrectionLevel: "M" });

  return Array.from({ length: modules.size }, (_, row) => Array.from({ length: modules.size }, (_, col) => Boolean(modules.get(row, col))));
}
```

```ts
// src/features/shops/shop-link.ts
import { shopUrl } from "./resolve-slug";

type Location = { hostname: string; port: string; protocol: string };

// The general link of a business: what customers type, scan or receive. The QR carries `url`, the screen prints `display`.
export function shopLink(slug: string, baseDomain: string, location?: Location): { display: string; url: string } {
  const url = location ? shopUrl(slug, baseDomain, location) : `https://${slug}.${baseDomain}/`;

  return { display: url.replace(/^https?:\/\//, "").replace(/\/$/, ""), url };
}
```

```tsx
// src/features/shops/QrCode.tsx
import Svg, { Path, Rect } from "react-native-svg";

const QUIET_ZONE = 4;

export function qrPath(matrix: boolean[][]): string {
  return matrix
    .flatMap((row, r) => row.flatMap((dark, c) => (dark ? [`M${c + QUIET_ZONE},${r + QUIET_ZONE}h1v1h-1z`] : [])))
    .join("");
}

// Black on white whatever the theme: a QR must keep its contrast to be scanned.
export function QrCode({ accessibilityLabel, matrix, size }: { accessibilityLabel: string; matrix: boolean[][]; size: number }) {
  const side = matrix.length + QUIET_ZONE * 2;

  return (
    <Svg accessibilityLabel={accessibilityLabel} accessibilityRole="image" height={size} testID="shop-link-qr" viewBox={`0 0 ${side} ${side}`} width={size}>
      <Rect fill="#FFFFFF" height={side} width={side} x={0} y={0} />
      <Path d={qrPath(matrix)} fill="#000000" shapeRendering="crispEdges" />
    </Svg>
  );
}
```

```ts
// src/features/shops/share-link.ts
export type ShareResult = "shared" | "copied" | "cancelled" | "failed";
export type Clipboard = { writeText(text: string): Promise<void> };
export type ShareDeps = {
  clipboard?: Clipboard;
  nativeShare?: (options: { message: string }) => Promise<unknown>;
  webShare?: (data: { title: string; url: string }) => Promise<void>;
};

const QUIET_ZONE = 4;

export async function copyLink(url: string, clipboard?: Clipboard): Promise<boolean> {
  if (!clipboard) return false;

  try {
    await clipboard.writeText(url);
    return true;
  } catch {
    return false;
  }
}

// Native: the system share sheet. Web: navigator.share when the browser has it (phones), otherwise copy the link.
export async function shareLink(url: string, title: string, deps: ShareDeps): Promise<ShareResult> {
  try {
    if (deps.nativeShare) {
      await deps.nativeShare({ message: url });
      return "shared";
    }
    if (deps.webShare) {
      await deps.webShare({ title, url });
      return "shared";
    }
  } catch (error) {
    if (error instanceof Error && error.name === "AbortError") return "cancelled";
  }

  return (await copyLink(url, deps.clipboard)) ? "copied" : "failed";
}

export function drawQr(ctx: Pick<CanvasRenderingContext2D, "fillRect" | "fillStyle">, matrix: boolean[][], scale: number) {
  const side = (matrix.length + QUIET_ZONE * 2) * scale;

  ctx.fillStyle = "#FFFFFF";
  ctx.fillRect(0, 0, side, side);
  ctx.fillStyle = "#000000";
  matrix.forEach((row, r) =>
    row.forEach((dark, c) => {
      if (dark) ctx.fillRect((c + QUIET_ZONE) * scale, (r + QUIET_ZONE) * scale, scale, scale);
    }),
  );
}

type DownloadDeps = { createCanvas?: () => HTMLCanvasElement; download?: (href: string, filename: string) => void };

const defaultCanvas = () => document.createElement("canvas");
const defaultDownload = (href: string, filename: string) => {
  const link = document.createElement("a");

  link.href = href;
  link.download = filename;
  link.click();
};

// Web only: draws the matrix on a canvas (10 px per module) and downloads it as a PNG.
export async function downloadQrPng(matrix: boolean[][], filename: string, deps: DownloadDeps = {}): Promise<void> {
  const canvas = (deps.createCanvas ?? defaultCanvas)();
  const scale = 10;
  const side = (matrix.length + QUIET_ZONE * 2) * scale;
  const ctx = canvas.getContext("2d");

  if (!ctx) throw new Error("Canvas is not available");
  canvas.width = side;
  canvas.height = side;
  drawQr(ctx, matrix, scale);
  (deps.download ?? defaultDownload)(canvas.toDataURL("image/png"), filename);
}
```

```tsx
import { useMemo, useState } from "react";
import { useTranslation } from "react-i18next";
import { Platform, Share, Text, View } from "react-native";

import { Toast } from "../../components/domain/Toast";
import { Button } from "../../components/ui/Button";
import { Card } from "../../components/ui/Card";
import { shopBaseDomain } from "./host";
import { useMyShop } from "./my-shop";
import { QrCode } from "./QrCode";
import { qrMatrix } from "./qr";
import { copyLink, downloadQrPng, shareLink } from "./share-link";
import { shopLink } from "./shop-link";

const isWeb = Platform.OS === "web";

// The business's general link with its QR code. The same component serves the owner, the professional and the last setup step.
export function ShopLinkCard({ audience }: { audience: "owner" | "professional" }) {
  const { t } = useTranslation();
  const mine = useMyShop();
  const slug = mine.data?.slug;
  const link = useMemo(() => (slug ? shopLink(slug, shopBaseDomain(), isWeb && typeof window !== "undefined" ? window.location : undefined) : null), [slug]);
  const matrix = useMemo(() => (link ? qrMatrix(link.url) : null), [link]);
  const [feedback, setFeedback] = useState<{ message: string; variant: "error" | "success" } | null>(null);

  if (!link || !matrix) return null;

  const clipboard = isWeb && typeof navigator !== "undefined" ? navigator.clipboard : undefined;
  const share = async () => {
    const result = await shareLink(link.url, link.display, {
      clipboard,
      nativeShare: isWeb ? undefined : (options) => Share.share(options),
      webShare: isWeb && typeof navigator !== "undefined" && typeof navigator.share === "function" ? (data) => navigator.share(data) : undefined,
    });

    if (result === "copied") setFeedback({ message: t("shopLinkScreen.copied"), variant: "success" });
    if (result === "failed") setFeedback({ message: t("shopLinkScreen.failed"), variant: "error" });
  };
  const copy = async () => setFeedback((await copyLink(link.url, clipboard)) ? { message: t("shopLinkScreen.copied"), variant: "success" } : { message: t("shopLinkScreen.failed"), variant: "error" });
  const download = () => downloadQrPng(matrix, `qr-${slug}.png`).catch(() => setFeedback({ message: t("shopLinkScreen.downloadFailed"), variant: "error" }));

  return (
    <View className="gap-4">
      <Text className="text-base font-sans text-neutral-700">{t(audience === "owner" ? "shopLinkScreen.introOwner" : "shopLinkScreen.introProfessional")}</Text>
      <Card className="items-center gap-4">
        <QrCode accessibilityLabel={t("shopLinkScreen.qrLabel")} matrix={matrix} size={240} />
        <Text className="text-center text-lg font-display-semibold text-ink" selectable testID="shop-link-url">{link.display}</Text>
      </Card>
      <Button label={t("shopLinkScreen.share")} onPress={() => void share()} testID="shop-link-share" />
      {isWeb ? <Button label={t("shopLinkScreen.copy")} onPress={() => void copy()} testID="shop-link-copy" variant="outline" /> : null}
      {isWeb ? <Button label={t("shopLinkScreen.download")} onPress={() => void download()} testID="shop-link-download" variant="outline" /> : null}
      <Toast message={feedback?.message ?? ""} onDismiss={() => setFeedback(null)} variant={feedback?.variant ?? "info"} visible={feedback !== null} />
    </View>
  );
}
```
